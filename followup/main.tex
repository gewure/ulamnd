\begin{abstract}
\status{draft; numbers to be filled from WP0--WP3}
Let $f\in\Z[t]$ be irreducible without fixed prime divisor, $K=\Q[t]/(f)$, and $S_f(h)$ its pair
singular series. In part I we proved that the diagonal part of $\sum_{h\le H}(S_f(h)-C(f)^2)$ has
Dirichlet series $D_f(s)=\zeta_K(s+1)E_f(s)$. Here we determine the analytic structure of $E_f$
completely: it is an infinite product of Artin $L$-functions of virtual representations of the Galois
group, the first of which is $L(2s+2,\Sym^2V_f)^{-1}=\bigl(\zeta_K(2s+2)\prod_O\zeta_{K_O}(2s+2)\bigr)^{-1}$
with $K_O$ the fields cut out by the unordered pairs of roots; $D_f$ is meromorphic in $\Re s>-1$, has
real poles at $s=-1+1/k$, and $\Re s=-1$ is a natural boundary. Consequently the Riesz means of the
diagonal satisfy an explicit formula, unconditional for $m\ge2$, whose oscillating part is indexed by the
zeros of these Dedekind zeta functions, with terms $x^{m-1+\rho/2}$, and whose smooth part contains
$x^{m-2/3}$. For $f=t^2+1$ every coefficient is computed exactly and the formula is verified against
the exact Riesz means to $x=4\cdot10^7$: \status{WP0 result}. We prove $\Omega_\pm(x^{m-3/4})$
unconditionally, $O(x^{m-7/8+\varepsilon})$ after the real terms under GRH for $\zeta_K$, and give the
function-field version, where the formula is an exact finite sum. Assuming the Hardy--Littlewood pair
conjecture with a power-saving error, these results describe the second-order term of the variance of
the number of prime values of $f$; we explain precisely why the zero-induced fluctuations lie below the
scale at which that transfer is valid, so that no statement about prime values themselves and the zeros
of $\zeta_K$ is claimed.
\end{abstract}

\section{The analytic structure of $D_f$}\label{sec:structure}

\subsection{The pair fields}

\begin{lemma}[pair fields]\label{lem:pairs}\status{PROVED}
Let $V_f$ be the permutation representation of $G=\operatorname{Gal}(f)$ on the roots. Then
$\Sym^2V_f\cong V_f\oplus V_f^{(2)}$, where $V_f^{(2)}$ is the permutation representation on unordered
pairs of distinct roots. Consequently
\[
L(s,\Sym^2V_f)=\zeta_K(s)\prod_{O}\zeta_{K_O}(s),
\]
the product over the $G$-orbits $O$ on unordered pairs, $K_O$ the fixed field of the stabiliser of a pair
in $O$. If $G$ is $2$-transitive there is one orbit and $K^{(2)}=\Q(\alpha+\beta,\alpha\beta)$ has degree
$n(n-1)/2$.
\end{lemma}
\begin{proof}
$\Sym^2V_f$ has the basis $e_\alpha e_\beta$, $\alpha\le\beta$, permuted by $G$; the multisets $\{\alpha,\alpha\}$
span a copy of $V_f$ and the remaining ones $V_f^{(2)}$. The $L$-function of a permutation representation
$\operatorname{Ind}_H^G\mathbf 1$ is the Dedekind zeta function of the fixed field of $H$.
\end{proof}
\status{Examples: quadratic: $\zeta^2L(\chi_D)$; $S_3$-cubic: $\zeta_K^2$; $S_4$-quartic: $\zeta_K\zeta_{K^{(2)}}$ with
$[K^{(2)}:\Q]=6$ (the resolvent sextic field); $A_4$: same degree $6$; $D_4$, $C_4$, $V_4$: two or three orbits.}

\subsection{The infinite product}

Write $u=p^{-1}$, $v=p^{-(s+1)}=p^{-w}$. For $p$ unramified in $K$ the local factor of $D_f/\zeta_K(w)$ is
\[
R_p(u,v)=\Bigl(P_p+p\,b(p)\,\omega_f(p)\,p^{-s}\Bigr)\det\bigl(1-\Frob_pv\mid V_f\bigr),\qquad
R_p(0,v)=\bigl(1+\omega(g)v\bigr)\prod_i\bigl(1-v^{\lambda_i(g)}\bigr),
\]
$g=\Frob_p$ of cycle type $\lambda(g)$ on the roots.

\begin{theorem}[structure of $E_f$]\label{thm:structure}\status{TO WRITE; proof is Möbius inversion + the
computation of the script for the mixed terms}
Let $c_k(g)$ be the coefficient of $v^k$ in $\log R(0,v)$ (a class function on $G$ with values in $\Q$), and
define the virtual characters $\Psi_k=\frac1k\sum_{j\mid k}\mu(k/j)\,j\,c_j(\cdot^{\,k/j})$ \status{fix the
exact inversion formula: it is the one solved level by level in \texttt{explicit-diag.py}; for $G$ acting on
two roots it gives the exponents of Table~\ref{tab:exponents}}. Then for every $K\ge2$
\[
E_f(s)=\prod_{k=2}^{K}L\bigl(k(s+1),\Psi_k\bigr)\cdot M_{f,K}(s),
\]
where $M_{f,K}$ is an Euler product absolutely convergent for $\Re s>-1+1/(K+1)$ \status{check the
exact abscissa: the pure-$v$ remainder is $O(v^{K+1})$ and the mixed terms are $O(p^{-2},p^{-2-w},p^{-1-2w})$}.
In particular $D_f$ is meromorphic in $\Re s>-1$, with poles only at $s=0$ and at the points where some
$L(k(s+1),\Psi_k)$, $k\ge2$, has a zero (if $\Psi_k$ has a negative multiplicity there) or a pole
(if positive), i.e. at $s=\rho/k-1$ and, for the trivial constituents, at $s=-1+1/k$.
\end{theorem}

\subsection{The natural boundary}

\begin{theorem}[natural boundary]\label{thm:boundary}\status{PROVED modulo writing}
$D_f$ has zeros at every $s$ with $p^{-s}=-P_p/(p\,b(p)\omega_f(p))$, $\omega_f(p)\ge1$, and these
accumulate at every point of the line $\Re s=-1$. Hence $\Re s=-1$ is a natural boundary of $D_f$, and
of $\sum_nA_f(n)n^{-s}=\zeta(s)D_f(s)$. The same holds for $f(t)=t$, i.e.\ for the generating series of
the twin-prime singular series.
\end{theorem}
\begin{proof}[Proof sketch]
In $\Re s>-1+1/(K+1)$ write $D_f=\zeta_K(w)\prod_{k\le K}L(kw,\Psi_k)M_{f,K}$; the absolutely convergent
product $M_{f,K}$ vanishes exactly where one of its local factors does, and the local factor at $p$
vanishes exactly where $F_p(s)=P_p+p\,b(p)\omega_f(p)p^{-s}=0$ (the other factors are non-zero for $\Re w>0$).
The zeros of $F_p$ have $\Re s=-\log\bigl(|P_p|/(p\,b(p)\omega_f(p))\bigr)/\log p\to-1$ and
$\Im s=(2j+1)\pi/\log p$, which is dense in $\mathbb R$ as $p\to\infty$. A meromorphic continuation to a
neighbourhood of a point $-1+it$ would have that point as an accumulation point of zeros, hence a pole,
near which there are no zeros: contradiction. \status{handle the finitely many ramified $p$ and the
poles of the $L$-factors; cite Estermann~\cite{Estermann1928}, Dahlquist~\cite{Dahlquist1952},
Bhowmik--Essouabri--Lichtin~\cite{BhowmikEssouabriLichtin2007} for context.}
\end{proof}

\section{The explicit formula for the diagonal}\label{sec:explicit}

Let $R_m(x)=\sum_{n\le x}(x-n)^mA_f(n)$ and $G_m(s)=m!\,\zeta(s)D_f(s)/\bigl(s(s+1)\cdots(s+m)\bigr)$, so that
$R_m(x)=\frac1{2\pi i}\int_{(2)}G_m(s)x^{s+m}\,ds$.

\begin{theorem}[quadratic case]\label{thm:explicit-quadratic}\status{TO PROVE: Goldston--Suriajaya's
contour argument with Theorem~\ref{thm:structure}; zero-density for $\zeta$ and $L(\chi_D)$}
Let $f$ be an irreducible quadratic of discriminant $D$, $m\ge2$. For every $\delta>0$,
\[
R_m(x)=\frac{D_f(1)}{m+1}x^{m+1}+x^m\Bigl(-\frac{\log x}{2C(f)}+c_f\Bigr)
+\sum_{\substack{k\ \mathrm{odd}\\3\le k<1/\delta}}x^{m-1+1/k}Q_{k}(\log x)
+x^{m-1}\sum_{\substack{k\ge2\\ k<1/(2\delta)}}\ \sum_{\substack{\rho_k\\|\gamma|\le U_k}}x^{\rho_k/k}Q_{k,\rho_k}(\log x)
+O_\delta(x^{m-1+\delta}),
\]
where $\rho_k$ runs over the zeros of $\zeta(\cdot)^{-a_k}L(\cdot,\chi_D)^{-b_k}$ (those with negative
exponent), $Q$ are polynomials of degree less than the order of the pole, and $U_k\asymp x^5$. The
$x^{m-2/3}$ term is present whenever $a_3>0$, i.e.\ for every quadratic.
\end{theorem}

Check for $S_n$, $A_n$; state the general case under Artin's conjecture + a zero-density hypothesis.}
\end{theorem}

\section{$\Omega$-results and conditional bounds}\label{sec:omega}

\begin{theorem}\status{TO PROVE: Landau's theorem as in GS Theorem 4; needs $A_f(n)\ge0$, true when
$P_p\ge0$ for all $p$, e.g.\ $t^2+1$; otherwise use the first zero directly as in GS JNT 2021}
For $f=t^2+1$ and every $m\ge1$, $R_m(x)-(\text{smooth terms})=\Omega_\pm(x^{m-3/4})$. Under RH for $\zeta$
and $L(s,\chi_{-4})$ and the simplicity of their zeros, the oscillating part is $O(x^{m-3/4+\varepsilon})$
and, for $m\ge2$ \status{and a Gonek--Hejhal-type bound}, an absolutely convergent sum
$x^{m-3/4}\sum_\gamma a_m(\rho)x^{i\gamma/2}$, with remainder $O(x^{m-3/4-\varepsilon_0})$ for some
$\varepsilon_0>0$. \status{Do NOT claim $O(x^{m-7/8+\varepsilon})$: the $k=4$ family at $\Re s=-7/8$ has
individually huge coefficients that cancel against the line integral ($|D_f|$ grows towards the natural
boundary); numerically it is rejected by the data. Remark on asymptotic vs convergent explicit formulas near
a natural boundary.}
\end{theorem}
