\begin{abstract}
Part~IV of this series~\cite{ReichardtIV} showed that the pieces $\mathcal P_u(Y)$ of the off-diagonal of the pair
singular series of an irreducible quadratic $f$ oscillate at the spectral parameters of the even Maass cusp forms, and
that the piece with dilation $u$ lives at level $u^2$. It left open ``the size, in $u$, of the restricted geometric
factor''. This note computes that factor for the level-one lines and for the newforms of level $u$. The piece with
dilation $u$ is the level-one seed of the smooth-window theorem, at length $uY$, summed over the $\Gamma_0(u^2)$-invariant
family $\mathcal W_u$ of forms of discriminant $4u^2D$ with $u^2\mid a$ and $2u^2\mid b$. The coefficient of the line $t_j$ of
a level-one form $u_j$ in piece $u$ is therefore a projection onto the oldform space $\{u_j(z),u_j(uz),u_j(u^2z)\}$, and
for an odd prime $u$ it equals the coefficient in piece $1$ times an explicit rational function of $\lambda_j(u)$,
$\sqrt u$ and $\chi_D(u)$: the Gram matrix of the oldforms (Iwaniec--Luo--Sarnak) and two local period identities for the
Hecke neighbours of Heegner points (Gross--Kohnen--Zagier). The decay in $u$ is set by the splitting type of $u$ in
$\Q(\sqrt D)$ and not by the Hecke eigenvalue: the ratio is exactly $u^{-3/2}$ when $u$ is inert, $-u^{-3/2}(1-2\lambda_j(u)u^{-1/2}+O(u^{-1}))$
when $u$ splits, and $u^{-1/2}(1-\lambda_j(u)u^{-1/2}+O(u^{-1}))$ when $u$ is ramified; the ratio is multiplicative in squarefree
$u$, and at $u=2$ the local constants depend on the class through the bilinear form modulo~$2$; so the main terms of the level-one lines
are uniformly summable over $u$. The same formula with a newform $v$ of level $u$ gives its line in piece $u$, and the
period that drives it vanishes when $u$ is inert, because the Hecke and Fricke contributions cancel, and for every $u$ when the
Fricke sign is $-1$. The piece splits into the $u=1$ object at length $uY$, a part of level $u$ and a part of genuinely level $u^2$;
this explains the inert identity, and in the data each named line lies in the part the splitting assigns it. Both predictions carry
no free parameter. Where the line is resolved in the data ($|r|\gtrsim0.1$ on grids to $Y=10^7$) they hold to a few percent at $u=2$ and to within the systematic uncertainty of the data, about $0.04$ in the ratio, at $u=3$,
and to a few hundredths of a radian in phase --- $13$ of the $21$ pairs $(u,D)$ tested for the first level-one line,
with one unexplained outlier, and five of the seven discriminants for the first newform of level~$3$, the other two being the
inert cases, where the line is observed at the noise level as predicted. The error term uniform in $u$, the open core of the
conjecture of part~III, is not touched.
\end{abstract}

\section{Introduction}\label{sec:intro}

\paragraph{The object.} Notation as in part~IV~\cite{ReichardtIV}. For a discriminant $D<0$, a squarefree $u\ge1$ and
$Q_u(h)=u^2h^2-D$, the unrestricted piece with dilation $u$ and smooth window $w\in C_c^\infty((0,\infty))$ is
\[
S^w_u(Y)=\sum_{h\ge1}w\Bigl(\frac hY\Bigr)\bigl(\sigma_{-1}(Q_u(h))-\mathbb E_u\bigr),\qquad
\sigma_{-1}(n)=\sum_{d\mid n}\frac1d,\quad \mathbb E_u=\sum_d\frac{\rho_u(d)}{d^2},
\]
$\rho_u(d)$ the number of roots of $u^2x^2\equiv D\pmod d$. For $u=1$ Theorem~\textsc{smooth} of part~IV gives
\begin{equation}\label{eq:smooth1}
\begin{gathered}
\sqrt Y\,S^w_1(Y)=|D|^{-1/4}\sum_j\Per_D(u_j)\,\tilde L_j\bigl[A_j(Y)+\overline{A_j(Y)}\bigr]+\mathcal E^w(Y)+O(Y^{-2+\varepsilon}),\\
A_j(Y)=\Gamma(-it_j)W_c\bigl(\tfrac32+it_j\bigr)\Bigl(\frac{\pi\sqrt{|D|}}Y\Bigr)^{it_j},
\end{gathered}
\end{equation}
the sum over an orthonormal basis of even Maass cusp forms of $\mathrm{SL}_2(\Z)$, with $\Per_D(u_j)=\sum_Qu_j(z_Q)/|\Gamma_Q|$
over all classes of forms of discriminant $4D$, $\tilde L_j=\sum_ka_j(k)k^{-3/2}$, and $W_c$ the Mellin factor of the window.
Part~IV, Section~7, showed that piece $u$ lives at level $u^2$ and named the one missing ingredient: how the coefficients
depend on $u$. Its footnote on $u=3,5$ recorded that the level-one lines are weak there and that the lines of $\Gamma_0(9)$
take over.

\paragraph{Results.} Write $\lambda_j(u)$ for the Hecke eigenvalue ($T_uu_j=\lambda_j(u)u_j$, $a_j(1)=1$) and
$\chi=\chi_D(u)$ for the Kronecker symbol, so $\chi=0$ iff $u\mid D$.

\begin{theorem}[the spectral formula for piece $u$]\label{thm:pieceu}
Let $\mathcal W_u=\{[a,b,c]\ \text{of discriminant }4u^2D:\ u^2\mid a,\ 2u^2\mid b\}$ (all contents). Then $\mathcal W_u$ is
$\Gamma_0(u^2)$-invariant and, for every $\varepsilon>0$,
\[
\sqrt Y\,S^w_u(Y)=\sqrt u\,|D|^{-1/4}\sum_v\Per_{\mathcal W_u}(v)\,\tilde L_v\bigl[A_v(uY)+\overline{A_v(uY)}\bigr]+\mathcal E^w_u(Y)+O_{w,D,u,\varepsilon}(Y^{-2+\varepsilon}),
\]
the sum over an orthonormal basis $\{v\}$ of the even Maass cusp forms of $\Gamma_0(u^2)$, with $\Per_{\mathcal W_u}(v)=
\sum_{Q\in\Gamma_0(u^2)\backslash\mathcal W_u}v(z_Q)/|\Gamma_0(u^2)_{z_Q}|$, $\tilde L_v$ formed with the Fourier coefficients
of $v$ at the cusp $\infty$, and $\mathcal E^w_u$ the continuous spectrum, $\mathcal E^w_u(Y)\ll_{A,u}Y^{-1/2}(\log Y)^{-A}$. Every
implied constant depends on $u$ (through the spectral data of $\Gamma_0(u^2)$); nothing here is uniform in $u$, which is why the
$u$-dependence of the coefficients is computed explicitly below.
\end{theorem}
\status{proved (Section~\ref{sec:seed}): it is Theorem~\textsc{smooth} of part~IV at level $u^2$ applied to an explicit
identity between seeds; every constant checked numerically at $u=2,3,5$.}

\begin{theorem}[the level-one lines]\label{thm:level1}
Let $u$ be an odd prime with $u^2\nmid D$ and $u_j$ an even Hecke--Maass form of level one. The coefficient of the line $t_j$ in
$\sqrt Y S^w_u(Y)$ equals the coefficient in $\sqrt YS^w_1(Y)$ multiplied by $u^{-it_j}$ and by
\begin{equation}\label{eq:r}
r_j(u;D)=\sqrt u\,P^{\mathsf T}\bigl(u(u+1)\,G\bigr)^{-1}L,\qquad
G=\begin{pmatrix}1&g&h\\ g&1&g\\ h&g&1\end{pmatrix},
\end{equation}
\[
g=\frac{\lambda_j(u)\sqrt u}{u+1},\qquad h=\frac{\lambda_j(u)^2-1-1/u}{u+1},\qquad L=(1,u^{-1},u^{-2}),\qquad P=(p_0,p_1,p_0),
\]
\[
p_1=\begin{cases}2u&u\parallel D,\\ u-\chi&u\nmid D,\end{cases}\qquad
p_0=\lambda_j(u)\sqrt u+\begin{cases}u-1&u\parallel D,\\ -(1+\chi)&u\nmid D,\end{cases}
\]
where $u\parallel D$ means $u\mid D$, $u^2\nmid D$. (For $u^2\mid D$, and for $u=2$ with every $D$, the entries of $P$ depend on the
class and are given by Proposition~\ref{prop:classwise}; the rest of~\eqref{eq:r} is unchanged.)
Moreover
\[
\begin{aligned}
r_j(u;D)&=u^{-3/2}&&\text{(inert),}\\
r_j(u;D)&=-u^{-3/2}\bigl(1-2\lambda_j(u)u^{-1/2}+O(u^{-1})\bigr)&&\text{(split),}\\
r_j(u;D)&=u^{-1/2}\bigl(1-\lambda_j(u)u^{-1/2}+(\lambda_j(u)^2+1)u^{-1}+O(u^{-3/2})\bigr)&&\text{(ramified),}
\end{aligned}
\]
the implied constants absolute, and the inert identity exact for every $\lambda_j(u)$. The Hecke eigenvalue enters only the corrections; the decay is set by the splitting type.
\end{theorem}
\status{proved (Section~\ref{sec:lemmas}); the asymptotics are read off~\eqref{eq:r} as a rational function of $\lambda$ and $\sqrt u$.
The inert identity has a one-line reason: for $\chi=-1$, $P=(\lambda\sqrt u,\,u+1,\,\lambda\sqrt u)=(u+1)(g,1,g)$ is $(u+1)$ times the
middle column of $G$, so $G^{-1}P=(u+1)e_2$ and $P^{\mathsf T}G^{-1}L=(u+1)/u$, whence $r=\sqrt u\,(u+1)/(u\cdot u(u+1))=u^{-3/2}$. A reason that uses no Gram matrix is Remark~\ref{rem:components}: for inert $u$ the
level-one lines of piece $u$ are exactly those of $\bar w\,S^w_1(uY)$ with $\bar w=1/u$. The first version of this draft stated the asymptotic as
$\lambda_j(u)/u$ for $u\nmid D$; that was wrong --- the $\lambda_j(u)\sqrt u$ terms of $p_0$ and $p_1L_2$ cancel exactly --- and was
caught by an outside reading that evaluated~\eqref{eq:r} at large $u$. The values of~\eqref{eq:r} agree with the direct
spectral computation to four digits and with the data to the noise (Table~\ref{tab:ratios}).}

\begin{theorem}[the newform lines]\label{thm:newform}
Let $u$ be an odd prime and $v$ an even Hecke--Maass newform of level $u$ with Atkin--Lehner eigenvalue $\varepsilon$
($a_v(u)=-\varepsilon u^{-1/2}$). Its line in $\sqrt YS^w_u(Y)$ is
\[
\begin{gathered}
\sqrt u\,|D|^{-1/4}\Bigl[\sum_{k=1}^2\Per_{\mathcal W_u}(v_k)\tilde L(v_k)\Bigr]\bigl[A_v(uY)+\overline{A_v(uY)}\bigr],\\
A_v(uY)=\Gamma(-it_v)W_c\bigl(\tfrac32+it_v\bigr)\Bigl(\frac{\pi\sqrt{|D|}}{uY}\Bigr)^{it_v},
\end{gathered}
\]
i.e.\ an amplitude $2\sqrt u|D|^{-1/4}|\sum_k\Per(v_k)\tilde L(v_k)|\,|\Gamma(-it_v)W_c(\tfrac32+it_v)|$ and a phase shifted by $-t_v\log u$;
$\{v_1,v_2\}$ the orthonormalisation of $\{v(z),v(uz)\}$ on $\Gamma_0(u^2)\backslash\HH$, whose Gram matrix has off-diagonal
entry $a_v(u)\sqrt u/u=-\varepsilon/u$ relative to $\langle v,v\rangle$. If $u$ is inert in $\Q(\sqrt D)$, or if $\varepsilon=-1$, then
$\Per_{\mathcal W_u}(v)=\Per_{\mathcal W_u}(v(u\,\cdot))=0$ and the line is absent.
\end{theorem}
\status{proved: the formula is Theorem~\ref{thm:pieceu} restricted to the two-dimensional oldspace, the Gram entry is
Lemma~\ref{lem:gram}(iii), and the vanishing is Lemma~\ref{lem:newperiod}. Tested on seven discriminants (Table~\ref{tab:newform}):
the period vanishes identically for inert $u$ in the computation, and in the data the line is at the noise level there. At $u=5$
the two even newforms with $\varepsilon=-1$ below $R=8.6$ are absent from the data and the three with the largest lines have
$\varepsilon=+1$ (Section~\ref{sec:numerics}).}

\begin{corollary}[uniform summability of the level-one main terms]\label{cor:sum}
In Theorem~A$'$ of part~III the piece with dilation $u$ enters at length $H/u$, so its level-one main terms are of size
$(H/u)^{1/2}|r_j(u;D)|$ times the $u=1$ amplitude. For a fixed level-one form $u_j$ and bounded weights $w(u)$,
$\sum_{u}|w(u)|\,(H/u)^{1/2}\,|r_j(u;D)|\ll_{D,\varepsilon} H^{1/2}$ over the squarefree $u$ with $p^2\nmid D$ for the odd $p\mid u$: by
Proposition~\ref{prop:mult} and Theorem~\ref{thm:level1}, $|r_j(u;D)|\le\prod_{p\mid u}|r_j(p;D)|\ll_\varepsilon u^{-3/2+\varepsilon}\prod_{p\mid(u,D)}p$, and
$\sum_uu^{-2+\varepsilon}$ converges. (The weights of part~III are
$w(u)=\lambda(u)\omega(u)\ll u^{\varepsilon}$; the conclusion is unchanged.)
\end{corollary}

\begin{remark}[what this does and does not settle]
Part~III's conjecture needs the pieces with $u\le H^{2/3}$ to sum to $o(H\log H)$ with their \emph{errors}. The
corollary says the level-one \emph{main} terms do so with room to spare; the newform lines of level $u$ are new at every
$u$ (Theorem~\ref{thm:newform} gives their size for each $u$ but not a bound uniform in the newform), the cycloidal-group
forms of level $u^2$~\cite{Stromberg2012} are not covered at all, and the error term of Remark~\textsc{gaps} of part~IV is
untouched. What has changed is that the main-term side of the $u$-problem is Hecke theory on Heegner divisors, and it
decays.
\end{remark}

\section{The seed identity and the family $\mathcal W_u$}\label{sec:seed}

The pairs $(d,x)$ with $u^2x^2\equiv D\pmod d$ are the forms $Q=[d,\,2ux,\,(u^2x^2-D)/d]$ of discriminant $4D$ with
$2u\mid b$; write $\mathcal H^{(u)}$ for this sub-family of the set $\mathcal H$ of all forms of discriminant $4D$, and
$z_Q=(-ux+i\sqrt{|D|})/d$. Poisson summation in $h$, exactly as in the proof of Theorem~\textsc{smooth}, writes
$S^w_u(Y)$ as $\sum_{(d,x)}\frac{Y}{d^2}\sum_{k\ne0}\hat w(kY/d)\,\e(kx/d)$, and in the coordinates of $z=z_Q$ ($y=\sqrt{|D|}/d$,
$x/d=-\Re z/u$) the summand is
\[
\frac{Yy^2}{|D|}\sum_{k\ne0}\hat w\Bigl(\frac{kYy}{\sqrt{|D|}}\Bigr)\e\Bigl(-\frac{k\Re z}u\Bigr)=u\,\Psi^w_{uY}(z/u),
\]
$\Psi^w_{Y'}(\tau)=\frac{Y'(\Im\tau)^2}{|D|}\sum_{k\ne0}\hat w(kY'\Im\tau/\sqrt{|D|})\e(-k\Re\tau)$ the level-one seed of
part~IV at length $Y'$. The point $\tau=z_Q/u=(-2u^2x+i\sqrt{4u^2|D|})/(2u^2d)$ is the Heegner point of the form
$Q'=[u^2d,\,2u^2x,\,(u^2x^2-D)/d]$ of discriminant $4u^2D$, and $Q\mapsto Q'$ is a bijection of $\mathcal H^{(u)}$ onto
$\mathcal W_u$. For $\gamma\in\Gamma_0(u^2)$ the middle coefficient transforms as $b'=2a\alpha\beta+b(\alpha\delta+\beta\gamma)+2c\gamma\delta
\equiv b\,\alpha\delta\equiv b\pmod{2u^2}$, since $u^2\mid a$, $u^2\mid\gamma$ and $\alpha\delta\equiv1\pmod{u^2}$; the
condition $u^2\mid a$ is $\Gamma_0(u^2)$-stable by Lemma~\textsc{subfamily} of part~IV. So $\mathcal W_u$ is a finite union of
$\Gamma_0(u^2)$-orbits, the translation $\tau\mapsto\tau+1$ is $x\mapsto x-d$, and $S^w_u(Y)=u\sum_{\tau\in\mathcal W_u/\Z}\Psi^w_{uY}(\tau)$.
The proof of Theorem~\textsc{smooth} applies verbatim on $\Gamma_0(u^2)\backslash\HH$ (its cusp at $\infty$ has width one, so
the pairing with the Fourier expansions at $\infty$ is unchanged), with $Y$ replaced by $uY$; the factor $u$ and
$(uY)^{-1/2}$ combine to $\sqrt u\,Y^{-1/2}$, and $(\pi\sqrt{|D|}/(uY))^{it_v}$ carries the phase $-t_v\log u$. This proves
Theorem~\ref{thm:pieceu}. For $f_m(z)=u_j(u^mz)$ the coefficient at index $u^mn$ is $u^{m/2}a_j(n)$, so $\tilde L_{f_m}=u^{-m}\tilde L_j$,
which is the vector $L$ of Theorem~\ref{thm:level1}.

\section{The three lemmas}\label{sec:lemmas}

Throughout $u$ is an odd prime and $\Per_D(f)=\sum_{Q\in\mathcal H/\mathrm{SL}_2(\Z)}f(z_Q)/|\Gamma_Q|$.

\begin{lemma}[multiplicity]\label{lem:mult}
Let $u$ be an odd prime with $u^2\nmid D$ (for odd $u$, $u\parallel D$ already forces $u$ prime to the conductor of the order of
discriminant $4D$). For every
$\mathrm{SL}_2(\Z)$-class $C$ of forms of discriminant $4D$,
$\sum_{Q'\in\Gamma_0(u^2)\backslash\mathcal W_u,\ uz_{Q'}\in C}|\Gamma_0(u^2)_{z_{Q'}}|^{-1}=m_u(D)\,|\Gamma_C|^{-1}$ with
$m_u(D)=2u$ if $u\parallel D$ and $m_u(D)=u-\chi_D(u)$ if $u\nmid D$. Equivalently $\Per_{\mathcal W_u}(u_j(u\,\cdot))=m_u(D)\Per_D(u_j)$.
Stabilisers are taken in $\mathrm{PSL}_2$ throughout (sizes $1$, $2$, $3$); since $-I$ lies in every group considered, the
convention only has to be the same on both sides of each identity.
\end{lemma}
\begin{proof}
Write $(Q\circ\gamma)(v)=Q(\gamma v)$ (a right action), $C=Q_0\circ\Gamma$ with $\Gamma=\mathrm{SL}_2(\Z)$, $S=\{\gamma:Q_0\circ\gamma=Q_0\}$,
and let $B$ be the bilinear form of $Q_0$, $B(v,w)=av_1w_1+\tfrac b2(v_1w_2+v_2w_1)+cv_2w_2$, of discriminant $D$.

\emph{The transported group.} With $\alpha=\left(\begin{smallmatrix}1&0\\0&u\end{smallmatrix}\right)$, $z\mapsto z/u$, one has
$\alpha^{-1}\Gamma_0(u^2)\alpha=G_u:=\Gamma_0(u)\cap\Gamma^0(u)=\{u\mid b,\ u\mid c\}$, of index $u(u+1)$. The bijection
$Q\mapsto Q'=[u^2a,ub,c]$ of $\mathcal H^{(u)}$ onto $\mathcal W_u$ has $z_{Q'}=z_Q/u$ and intertwines the action of $G_u$ on
$\mathcal H^{(u)}$ with that of $\Gamma_0(u^2)$ on $\mathcal W_u$, orbits and stabilisers included; and $\mathcal H^{(u)}$ is
$G_u$-stable, since $b(Q\circ g)\equiv b\,\alpha\delta\equiv b\pmod u$ for $g\in G_u$. So we count $G_u$-orbits in
$C\cap\mathcal H^{(u)}$.

\emph{Cosets are ordered pairs of lines.} $\Gamma\to\mathrm{SL}_2(\mathbb F_u)$ is onto and $G_u$ is the preimage of the diagonal
torus $T$, so $\Gamma/G_u\cong\mathrm{SL}_2(\mathbb F_u)/T$: a matrix with columns $(v,w)$ modulo $(v,w)\mapsto(\lambda v,\lambda^{-1}w)$
is the ordered pair of distinct lines $(\ell_1,\ell_2)=(\langle v\rangle,\langle w\rangle)$ in $\mathbb P^1(\mathbb F_u)$,
$u(u+1)$ pairs in all.

\emph{The family condition is orthogonality.} The middle coefficient of $Q_0\circ\gamma$ is $2B(\gamma e_1,\gamma e_2)$; so
$Q_0\circ\gamma\in\mathcal H^{(u)}$ iff $\ell_1\perp_B\ell_2$, a condition on the right coset $\gamma G_u$ and unchanged under
$\gamma\mapsto s\gamma$, $s\in S$.

\emph{The mass formula.} $C\cong S\backslash\Gamma$, the $G_u$-orbits in $C$ are the double cosets $S\gamma G_u$, with stabiliser
$G_u\cap\gamma^{-1}S\gamma$, and each double coset consists of $|S|/|G_u\cap\gamma^{-1}S\gamma|$ right cosets $\gamma'G_u$. Hence for
every function $\varphi$ on $\Gamma$ that is left-$S$- and right-$G_u$-invariant,
\begin{equation}\label{eq:mass}
\sum_{\text{$G_u$-orbits }O\subset C}\frac{\varphi(O)}{|\mathrm{Stab}_{G_u}(O)|}=\frac1{|S|}\sum_{\gamma\in\Gamma/G_u}\varphi(\gamma).
\end{equation}
With $\varphi=\mathbf 1[\ell_1\perp\ell_2]$ this gives $m_u(D)=\#\{(\ell_1,\ell_2)\text{ distinct}:\ \ell_1\perp_B\ell_2\}$. If
$u\nmid D$, $B$ is nondegenerate modulo $u$; every $\ell_1$ has a unique $\ell_1^\perp$, admissible iff $\ell_1^\perp\ne\ell_1$, i.e.\ iff
$\ell_1$ is not isotropic; the isotropic lines are the roots of $Q_0$ modulo $u$, $1+\chi_D(u)$ of them; so $m=u+1-(1+\chi)=u-\chi$.
If $u\parallel D$, then $Q_0\not\equiv0\pmod u$ for every class (a content divisible by $u$ would force $u^2\mid D$) and $B$ has rank one modulo $u$ with radical line $r$
(the double root); $B(v,w)=0$ iff $\langle v\rangle=r$ or $\langle w\rangle=r$, giving $u+u=2u$ ordered pairs.
\end{proof}

\begin{lemma}[the period of the dilated family]\label{lem:period}
Under the hypotheses of Lemma~\ref{lem:mult},
$\Per_{\mathcal W_u}(u_j)=\Per_{\mathcal W_u}(u_j(u^2\,\cdot))=\bigl(\lambda_j(u)\sqrt u+c_u(D)\bigr)\Per_D(u_j)$ with
$c_u(D)=u-1$ if $u\parallel D$ and $c_u(D)=-(1+\chi_D(u))$ if $u\nmid D$.
\end{lemma}
\begin{proof}
In the notation of the previous proof, $z_{Q_0\circ\gamma}=\gamma^{-1}z_0$, so $z_{Q_0\circ\gamma}/u=\alpha\gamma^{-1}z_0$ with
$\alpha\gamma^{-1}$ of determinant $u$. Since $\alpha M\alpha^{-1}=\left(\begin{smallmatrix}a&b/u\\ uc&d\end{smallmatrix}\right)$ is integral iff $u\mid b$, the coset
$\Gamma\alpha\gamma^{-1}$ depends on $\gamma$ only modulo right multiplication by $\Gamma^0(u)$, which fixes the second column
line $\ell_2=\langle\gamma e_2\rangle$ and moves the first; the map $\gamma\mapsto\Gamma\alpha\gamma^{-1}$ is a bijection of
$\Gamma/\Gamma^0(u)$ onto the $u+1$ cosets $\Gamma\backslash\Delta_u$ of the Hecke operator. (Check: $\gamma=\left(\begin{smallmatrix}1&-k\\0&1\end{smallmatrix}\right)$
has $\ell_1=\langle e_1\rangle$ fixed, $\ell_2=\langle(-k,1)\rangle$, and $\alpha\gamma^{-1}z=(z+k)/u$, $u$ distinct neighbours.)
Hence $u_j(\alpha\gamma^{-1}z_0)=:u_j(h_{\ell_2}(z_0))$ depends only on $\ell_2$, the points $h_\ell(z_0)$, $\ell\in\mathbb P^1(\mathbb F_u)$,
are the $u+1$ Hecke neighbours of $z_0$, and $\sum_\ell u_j(h_\ell(z_0))=\sqrt u\,\lambda_j(u)u_j(z_0)$. Applying~\eqref{eq:mass} with
$\varphi(\gamma)=u_j(\alpha\gamma^{-1}z_0)\mathbf 1[\ell_1\perp\ell_2]$ and summing over the classes,
\[
\Per_{\mathcal W_u}(u_j)=\sum_C\frac1{|S_C|}\sum_{\ell_2}N(\ell_2)\,u_j(h_{\ell_2}(z_C)),\qquad N(\ell_2)=\#\{\ell_1\ne\ell_2:\ \ell_1\perp\ell_2\}.
\]
If $u\nmid D$ then $N(\ell_2)=1$ for non-isotropic and $0$ for isotropic $\ell_2$, so the inner sum is
$\sqrt u\lambda_j(u)u_j(z_C)-\sum_{\ell\text{ isotropic}}u_j(h_\ell(z_C))$. The isotropic lines are the eigenlines of the order
$\mathcal O_{4D}$ acting on $L_C/uL_C$, $L_C=\Z+\Z z_C$, i.e.\ the $1+\chi$ sublattices $\mathfrak pL_C$ for the primes
$\mathfrak p\mid u$ of the order; their Heegner points have discriminant $4D$ again, with class $[\mathfrak p]\cdot C$
\cite[II]{GrossKohnenZagier1987}. Since $C\mapsto\mathfrak pC$ permutes the classes of each order and preserves $|S_C|$ (the
units of the order), $\sum_C|S_C|^{-1}\sum_{\text{iso}}u_j(h_\ell(z_C))=(1+\chi)\Per_D(u_j)$, and the claim follows.
If $u\parallel D$, $N(r)=u$ and $N(\ell_2)=1$ for $\ell_2\ne r$, so the inner sum is
$\sqrt u\lambda_j(u)u_j(z_C)+(u-1)u_j(h_r(z_C))$, and $h_r(z_C)=z_{\mathfrak pC}$ for the ramified prime $\mathfrak p$, again a
permutation of the classes: $\Per_{\mathcal W_u}(u_j)=(\sqrt u\lambda_j(u)+u-1)\Per_D(u_j)$. Finally
$u^2z_{Q'}=u z_Q=\alpha'\gamma^{-1}z_0$ with $\alpha'=\left(\begin{smallmatrix}u&0\\0&1\end{smallmatrix}\right)$, whose Hecke coset is
determined by $\ell_1$; the counts $\#\{\ell_2\ne\ell_1:\ell_1\perp\ell_2\}$ equal the $N(\ell_2)$ by the symmetry of $B$, which gives
$\Per_{\mathcal W_u}(u_j(u^2\cdot))=\Per_{\mathcal W_u}(u_j)$. (The same equality follows from the Fricke involution $\tau\mapsto-1/(u^2\tau)$
of $\Gamma_0(u^2)$, which maps $[u^2a',2u^2b',c]$ to $[u^2c,-2u^2b',a']$ --- the pairing of the divisor $d$ with $Q_u(x)/d$ --- and
so permutes $\mathcal W_u$.)
\end{proof}

\begin{proposition}[the class-by-class form of Lemmas~\ref{lem:mult} and~\ref{lem:period}]\label{prop:classwise}
Let $u$ be any prime and $D<0$ any discriminant (no hypothesis on $u$ and $D$). For a class $C$ of forms of discriminant $4D$
with representative $Q_C=[a,b,c]$ and bilinear form $B_C$, let
\begin{gather*}
m(C)=\#\{(\ell_1,\ell_2)\ \text{distinct}:\ B_C(\ell_1,\ell_2)\equiv0\ (u)\},\\
N_C(\ell_2)=\#\{\ell_1\ne\ell_2:\ B_C(\ell_1,\ell_2)\equiv0\ (u)\},
\end{gather*}
over the lines of $\mathbb P^1(\mathbb F_u)$, and let $h_\ell$ be the Hecke neighbour fixed by $\ell$: $h_{\langle(k,1)\rangle}z=(z-k)/u$,
$h_{\langle(1,0)\rangle}z=-1/(uz)$. Then
\begin{gather*}
\Per_{\mathcal W_u}(u_j(u\,\cdot))=\sum_C\frac{m(C)}{|S_C|}u_j(z_C),\\
\Per_{\mathcal W_u}(u_j)=\Per_{\mathcal W_u}(u_j(u^2\,\cdot))=\sum_C\frac1{|S_C|}\sum_{\ell}N_C(\ell)\,u_j(h_\ell z_C),
\end{gather*}
and Theorem~\ref{thm:level1} holds with $p_1=\Per_{\mathcal W_u}(u_j(u\cdot))/\Per_D(u_j)$, $p_0=\Per_{\mathcal W_u}(u_j)/\Per_D(u_j)$.
For $u=2$ the counts depend only on $(a,b/2,c)$ modulo $2$: with $b'=b/2$,
\begin{gather*}
m(C)=2\bigl([b'\equiv0]+[a\equiv b']+[c\equiv b']\bigr),\qquad N_C(\langle e_2\rangle)=[b'\equiv0]+[c\equiv b'],\\
N_C(\langle e_1+e_2\rangle)=[a\equiv b']+[c\equiv b'],\qquad N_C(\langle e_1\rangle)=[b'\equiv0]+[a\equiv b'];
\end{gather*}
so $m(C)=4$ for a primitive class of even $D$, $6$ for a class of even content, $2$ for a class of odd $D$ that is not of even content,
and $0$ for a class of even content and odd $D$ (where $B_C$ is alternating modulo $2$).
\end{proposition}
\begin{proof}
Steps 1--4 of the proof of Lemma~\ref{lem:mult} and the first half of the proof of Lemma~\ref{lem:period} (the coset bijection,
the orthogonality condition, the mass formula~\eqref{eq:mass} and the identification of $\alpha\gamma^{-1}z_0$ with the
neighbour $h_{\ell_2}z_0$) use neither the parity of $u$ nor any hypothesis on $D$; the two displays are~\eqref{eq:mass} with
$\varphi=\mathbf 1[\ell_1\perp\ell_2]$ and $\varphi=u_j(\alpha\gamma^{-1}z_0)\mathbf 1[\ell_1\perp\ell_2]$, and the equality of the
periods of $f_0$ and $f_2$ is the symmetry of $B_C$ as before. Only the evaluation of the counts used the hypotheses: for odd $u$
with $u^2\nmid D$ the reduction $B_C\bmod u$ is determined by $D$ up to equivalence, hence the counts are class-independent;
for $u=2$ the three lines are $\langle e_1\rangle,\langle e_2\rangle,\langle e_1+e_2\rangle$ with $B_C(e_1,e_2)=b'$,
$B_C(e_1,e_1+e_2)=a+b'$, $B_C(e_2,e_1+e_2)=b'+c$, which gives the table (a nondegenerate symmetric form over $\mathbb F_2$ is
not determined by its determinant: $\left(\begin{smallmatrix}1&0\\0&1\end{smallmatrix}\right)$ and
$\left(\begin{smallmatrix}0&1\\1&0\end{smallmatrix}\right)$ are inequivalent, which is why the counts depend on the class);
for $u^2\mid D$ the classes of content divisible by $u$ have $B_C\equiv0$ and $m(C)=u(u+1)$. The case distinction for $m(C)$
at $u=2$ is read off $b'^2-ac=D$: for even $D$, either $b'$ is odd and $a,c$ are odd, or $b'$ is even and exactly one of $a,c$ is even,
both giving $m=4$, or $a,b,c$ are all even ($m=6$); for odd $D$, either $b'$ is even and $a,c$ odd, or $b'$ odd and exactly one of $a,c$
even, both giving $m=2$, or $b'$ is odd and $a,c$ even, the even-content classes, where $B_C\bmod2$ is alternating and $m=0$.
\end{proof}
\begin{remark}\label{rem:classwise}
The two displays of Proposition~\ref{prop:classwise} were checked against the direct enumeration of the
$\Gamma_0(u^2)$-orbits of $\mathcal W_u$ (\texttt{hecke-classwise.py}) for $u=2$ and nine discriminants
$-3\ge D\ge-20$, and for $u=3$ with $D=-9,-27,-36$; the two routes agree to the four digits printed. At $u=2$ the resulting $p_1$ takes the values
$4.913$ ($D\equiv0\ (4)$, $D/4\equiv2,3\ (4)$), $5.278$ ($D/4\equiv1\ (4)$), $5.565$ ($D/4\equiv0\ (4)$), $0.321$ ($D\equiv1\ (8)$),
$1.373$ ($D\equiv5\ (8)$): by the table, $p_1=4+2\,\Per^{(2)}_D/\Per_D$ for even $D$ and $2-2\,\Per^{(2)}_D/\Per_D$ for odd $D$, where
$\Per^{(2)}_D$ is the period over the classes of even content, i.e.\ over the forms $2Q'$ with $Q'$ of discriminant $D$; the observed
values give $\Per_D/\Per^{(2)}_D=\sqrt2\lambda_j(2)-\chi_D(2)$ to four digits when $D$ is odd or $D/4\equiv2,3\ (4)$ (when $D/4$ is
not itself a discriminant), as the Hecke relation at $2$ between the periods of discriminants $D$ and $4D$ suggests, and a
different value when $D/4$ is a discriminant ($1.564$ at $D=-12$, $1.278$ at $D=-16$), where that relation has a further term. We have not used this. For $u=3$ and $9\mid D$ the ratio $r_1(3;D)$ exceeds $1$
($4.22$ at $D=-27$, $1.75$ at $D=-9,-36$) because the classes of content $3$ have $m(C)=12$.
\end{remark}

\begin{lemma}[the Gram matrix]\label{lem:gram}
On $\Gamma_0(u^2)\backslash\HH$, with $f_m(z)=u_j(u^mz)$: (i) $\langle f_0,f_0\rangle=u(u+1)\|u_j\|^2_{\mathrm{SL}_2(\Z)}$ and
$\langle f_1,f_0\rangle=\langle f_2,f_1\rangle=\frac{\lambda_j(u)\sqrt u}{u+1}\langle f_0,f_0\rangle$;
(ii) $\langle f_2,f_0\rangle=\frac{\lambda_j(u)^2-1-1/u}{u+1}\langle f_0,f_0\rangle$; (iii) for a newform $v$ of level $u$ with Fricke
eigenvalue $\varepsilon$, $\langle v(u\,\cdot),v\rangle=-\frac\varepsilon u\langle v,v\rangle$.
\end{lemma}
\begin{proof}
Write $\mathrm{Tr}_H^G F=\sum_{\gamma\in H\backslash G}F\circ\gamma$ for $H\subset G$ of finite index, so that
$\langle F,\Phi\rangle_H=\langle\mathrm{Tr}_H^GF,\Phi\rangle_G$ for $G$-invariant $\Phi$, and $\langle\Phi,\Psi\rangle_H=[G:H]\langle\Phi,\Psi\rangle_G$
for $G$-invariant $\Phi,\Psi$. (i) The cosets $\Gamma_0(u)\backslash\Gamma$ are $\left(\begin{smallmatrix}1&0\\ b&1\end{smallmatrix}\right)$,
$b\bmod u$, and $S$, and the points $u\gamma z$ over them are the $u+1$ Hecke neighbours of $z$; so
$\mathrm{Tr}_{\Gamma_0(u)}^\Gamma(u_j(u\,\cdot))=\sqrt u\lambda_j(u)u_j$, $\langle u_j(uz),u_j\rangle_{\Gamma_0(u)}=\sqrt u\lambda_j(u)\|u_j\|^2_\Gamma=
\frac{\sqrt u\lambda_j(u)}{u+1}\|u_j\|^2_{\Gamma_0(u)}$ \cite[Lemma~2.4]{IwaniecLuoSarnak2000}, and the ratio persists on $\Gamma_0(u^2)$
since both functions are $\Gamma_0(u)$-invariant; $\langle f_2,f_1\rangle$ reduces to the same by $z\mapsto z/u$, which carries
$\Gamma_0(u^2)\backslash\HH$ to $(\Gamma_0(u)\cap\Gamma^0(u))\backslash\HH$ of the same index. (ii) Trace from $\Gamma_0(u^2)$ to $\Gamma_0(u)$
over $\gamma_b=\left(\begin{smallmatrix}1&0\\ ub&1\end{smallmatrix}\right)$: $f_2(\gamma_bz)=u_j\bigl(\left(\begin{smallmatrix}u&0\\ b&1\end{smallmatrix}\right)uz\bigr)$,
and $\left(\begin{smallmatrix}u&0\\ b&1\end{smallmatrix}\right)\in\Gamma\left(\begin{smallmatrix}1&k\\0&u\end{smallmatrix}\right)$ with $k\equiv b^{-1}$ for
$b\ne0$, so $f_2(\gamma_bz)=u_j((uz+k)/u)$ for $b\ne0$ and $u_j(u^2z)$ for $b=0$; by the Hecke relation at the point $uz$,
$\mathrm{Tr}\,f_2=\sum_{k\bmod u}u_j((uz+k)/u)-u_j(z)+u_j(u^2z)=\sqrt u\lambda_j(u)u_j(uz)-u_j(z)$. Hence
$\langle f_2,f_0\rangle_{\Gamma_0(u^2)}=\sqrt u\lambda_j(u)\langle f_1,f_0\rangle_{\Gamma_0(u)}-\|u_j\|^2_{\Gamma_0(u)}
=\bigl(\tfrac{\lambda_j(u)^2u}{u+1}-1\bigr)\|u_j\|^2_{\Gamma_0(u)}=\tfrac{\lambda_j(u)^2-1-1/u}{u+1}\langle f_0,f_0\rangle$.
(iii) The same trace gives $\mathrm{Tr}\,v(u\,\cdot)=\sum_b v\bigl(\left(\begin{smallmatrix}1&0\\ b&1\end{smallmatrix}\right)uz\bigr)$, and
$\sum_bv\bigl(\left(\begin{smallmatrix}1&0\\ b&1\end{smallmatrix}\right)w\bigr)=-v(-1/w)$ because a newform has vanishing trace to
level one; with $w=uz$, $\mathrm{Tr}\,v(u\,\cdot)=-v(-1/(uz))=-\varepsilon v(z)$, so $\langle v(uz),v\rangle_{\Gamma_0(u^2)}=-\varepsilon\|v\|^2_{\Gamma_0(u)}=-\frac\varepsilon u\|v\|^2_{\Gamma_0(u^2)}$.
\end{proof}

\begin{proposition}[multiplicativity in $u$]\label{prop:mult}
Let $u$ be squarefree and $D<0$ with $p^2\nmid D$ for every odd prime $p\mid u$. Write $\{f_d(z)=u_j(dz):d\mid u^2\}$ for the
oldform space of $u_j$ at level $u^2$, $d=\prod_pp^{a_p}$ with $a_p\in\{0,1,2\}$, and $\mathbf P_d=\Per_{\mathcal W_u}(f_d)$,
$\mathbf G_{dd'}=\langle f_d,f_{d'}\rangle_{\Gamma_0(u^2)}$, $\mathbf L_d=\tilde L_{f_d}=\tilde L_j/d$. Then
\[
\frac{\mathbf P_d}{\Per_D(u_j)}=\prod_{p\mid u}p^{(p)}_{a_p},\qquad
\frac{\mathbf G_{dd'}}{\langle f_1,f_1\rangle}=\prod_{p\mid u}\hat G^{(p)}_{a_pa'_p},\qquad
\langle f_1,f_1\rangle=\prod_{p\mid u}p(p+1)\,\|u_j\|^2,
\]
where $(p^{(p)}_0,p^{(p)}_1,p^{(p)}_2)=(p_0,p_1,p_0)$ is the vector $P$ of Theorem~\ref{thm:level1} at the prime $p$ (for $p=2$
the class-by-class vector of Proposition~\ref{prop:classwise}) and $\hat G^{(p)}$ is the normalised $3\times3$ Gram matrix
of Lemma~\ref{lem:gram} at $p$. Consequently $r_j(u;D)=\prod_{p\mid u}r_j(p;D)$.
\end{proposition}
\begin{proof}
\emph{The periods.} Steps 1--4 of the proof of Lemma~\ref{lem:mult} hold verbatim for squarefree $u$: $G_u=\Gamma_0(u)\cap\Gamma^0(u)$
is the preimage of $\prod_{p\mid u}T_p$ under $\Gamma\to\prod_p\mathrm{SL}_2(\mathbb F_p)$, so a coset $\gamma G_u$ is a tuple
$(\ell_1^{(p)},\ell_2^{(p)})_{p\mid u}$ of ordered pairs of distinct lines, and the family condition $u\mid B(\gamma e_1,\gamma e_2)$
is $\ell_1^{(p)}\perp\ell_2^{(p)}$ for every $p$. For $d\mid u^2$ the point $d\cdot z_{Q_0\circ\gamma}/u$ is
$\mathrm{diag}(d,u)\gamma^{-1}z_0$, and the $p$-part of $\mathrm{diag}(d,u)$ is a scalar if $a_p=1$, $\mathrm{diag}(1,p)$ if $a_p=0$
and $\mathrm{diag}(p,1)$ if $a_p=2$; so the $\Gamma$-coset of $\mathrm{diag}(d,u)\gamma^{-1}$ is determined by the lines
$\ell_2^{(p)}$ ($a_p=0$) and $\ell_1^{(p)}$ ($a_p=2$) alone, and the point is the composite Hecke neighbour
$\prod_{a_p\ne1}h^{(p)}z_0$, the Heegner point of the sublattice of $L_C$ cut out by those lines (the $h^{(p)}$ commute:
sublattices of coprime index intersect). The mass formula~\eqref{eq:mass} thus gives
\[
\Per_{\mathcal W_u}(f_d)=\sum_C\frac1{|S_C|}\prod_{a_p=1}m_p(C)\ \sum_{(\ell^{(p)})_{a_p\ne1}}\prod_{a_p\ne1}N^{(p)}_C(\ell^{(p)})\;
u_j\Bigl(\prod_{a_p\ne1}h^{(p)}_{\ell^{(p)}}z_C\Bigr),
\]
with $m_p$, $N^{(p)}_C$ as in Proposition~\ref{prop:classwise}. For an odd $p\mid u$ the inner operator is, as in the proof of
Lemma~\ref{lem:period}, $\sqrt p\lambda_j(p)-\mathrm{Iso}_p$ ($p\nmid D$) or $\sqrt p\lambda_j(p)+(p-1)\mathrm{Ram}_p$ ($p\parallel D$),
where $\mathrm{Iso}_p$ and $\mathrm{Ram}_p$ replace $z_C$ by $z_{\mathfrak pC}$ for the primes $\mathfrak p\mid p$ of the order.
The operator $\sqrt p\lambda_j(p)$ (the Hecke operator $T_p$) commutes with everything; and $C\mapsto\mathfrak pC$ is a permutation of
the classes preserving $|S_C|$ under which every other prime's operator is transported: $(L_C/qL_C,B_C)\cong(\mathfrak pL_C/q\mathfrak pL_C,B_{\mathfrak pC})$
for $q\ne p$, since $[L_C:\mathfrak pL_C]=p$ is prime to $q$ and $Q_{\mathfrak pC}=Q_C/p$ with $p$ a unit modulo $q$; under this
isomorphism the index-$q$ sublattice $M\subset L_C$ corresponds to $\mathfrak pM=M\cap\mathfrak pL_C$, so
$N^{(q)}_{\mathfrak pC}(\mathfrak pM)=N^{(q)}_C(M)$ and $h^{(q)}_Mh^{(p)}_{\mathfrak p}z_C=z_{\mathfrak pM}=h^{(q)}_{\mathfrak pM}z_{\mathfrak pC}$.
Summing over $C$ one prime at a time therefore extracts the factor $p^{(p)}_{a_p}$ for each odd $p$, leaving
$\sum_C|S_C|^{-1}\,\mathrm{Op}^{(2)}_{C,a_2}u_j(z_C)=p^{(2)}_{a_2}\Per_D(u_j)$ if $u$ is even and $\Per_D(u_j)$ otherwise.
(Only one prime may carry a class-dependent operator; for squarefree $u$ under the hypothesis this is at most $p=2$.)

\emph{The Gram matrix.} For $a,b\ge1$ put $g=(a,b)$, $a'=a/g$, $b'=b/g$. Unfolding against the Eisenstein series of
$\Gamma_0(u^2)$ at $\infty$ (Rankin--Selberg), $\langle u_j(a\,\cdot),u_j(b\,\cdot)\rangle_{\Gamma_0(u^2)}$ is a constant depending
only on $u_j$ and $u^2$ times the residue at $s=1$ of $\sum_{a\mid m,\,b\mid m}\sqrt{ab}\,\lambda_j(m/a)\lambda_j(m/b)m^{-s}
=(a'b')^{-1/2}(ga'b')^{1-s}\sum_k\lambda_j(a'k)\lambda_j(b'k)k^{-s}$. Since $k\mapsto\lambda_j(a'k)\lambda_j(b'k)$ is multiplicative
and its Euler factors at $p\nmid a'b'$ are those of $\sum\lambda_j(k)^2k^{-s}$, the ratio of residues is
\[
\frac{\langle u_j(a\,\cdot),u_j(b\,\cdot)\rangle}{\langle u_j,u_j\rangle_{\Gamma_0(u^2)}}=(a'b')^{-1/2}\prod_{p\mid a'b'}R_p(v_p(a'b')),\qquad
R_p(k)=\frac{\sum_{e\ge0}\lambda_j(p^{k+e})\lambda_j(p^e)p^{-e}}{\sum_{e\ge0}\lambda_j(p^e)^2p^{-e}},
\]
the local series converging absolutely by the bound $|\lambda_j(p^e)|\le(e+1)p^{7e/64}$. With $a=d$, $b=d'$ this is
$\prod_pp^{-|a_p-a'_p|/2}R_p(|a_p-a'_p|)$, a tensor product of Toeplitz $3\times3$ matrices, whose entries must be those of
Lemma~\ref{lem:gram} (take $u=p$): $p^{-1/2}R_p(1)=\lambda_j(p)\sqrt p/(p+1)$ and $p^{-1}R_p(2)=(\lambda_j(p)^2-1-1/p)/(p+1)$. (Checked
numerically from the Hecke relations at $p=2,3,5$; \texttt{hecke-classwise.py}.) The normalisation is $[\Gamma:\Gamma_0(u^2)]=\prod p(p+1)$.

\emph{The ratio.} $\mathbf L=\bigotimes_p(1,p^{-1},p^{-2})\tilde L_j$, so $\sqrt u\,\mathbf P^{\mathsf T}\mathbf G^{-1}\mathbf L$ factors
as $\prod_p\sqrt p\,P^{(p)\mathsf T}(p(p+1)\hat G^{(p)})^{-1}L^{(p)}\cdot\Per_D(u_j)\tilde L_j/\|u_j\|^2$, and dividing by the $u=1$
coefficient gives $\prod_pr_j(p;D)$.
\end{proof}


\begin{lemma}[the newform periods]\label{lem:newperiod}
Let $v$ be a newform of level $u$ with Fricke eigenvalue $\varepsilon$, under the hypotheses of Lemma~\ref{lem:mult}. Then
$\Per_{\mathcal W_u}(v(u\,\cdot))=\varepsilon\Per_{\mathcal W_u}(v)$, and
\[
\Per_{\mathcal W_u}(v)=\begin{cases}0&\text{$u$ inert,}\\ -\sum_C|S_C|^{-1}\sum_{\ell_2\text{ isotropic}}v(h_{\ell_2}(z_C))&\text{$u$ split,}\\
(u-1)\sum_C|S_C|^{-1}v(h_r(z_C))&\text{$u\parallel D$,}\end{cases}
\]
where the $h_\ell(z_C)$ are taken as points of $X_0(u)$: the sums on the right are periods of $v$ over the Heegner points of
level $u$ and discriminant $4D$ above each class, in the sense of Gross--Kohnen--Zagier. In particular
$\Per_{\mathcal W_u}(v)=0$ whenever $\varepsilon=-1$, whatever the splitting of $u$.
\end{lemma}
\begin{proof}
For fixed $\ell_2$ the $u$ choices of $\ell_1$ are $\gamma\mapsto\gamma\left(\begin{smallmatrix}1&0\\ c&1\end{smallmatrix}\right)$, and
$\alpha\left(\begin{smallmatrix}1&0\\-c&1\end{smallmatrix}\right)\alpha^{-1}=\left(\begin{smallmatrix}1&0\\-uc&1\end{smallmatrix}\right)\in\Gamma_0(u)$;
so $v(\alpha\gamma^{-1}z_0)=:V(\ell_2)$ depends only on $\ell_2$ also for the level-$u$ form. As points of $X_0(u)$ the $u+1$
values are $(z_0+k)/u$ ($\ell_2=\langle(-k,1)\rangle$) and $-1/(uz_0)$ ($\ell_2=\langle e_1\rangle$), so
$\sum_{\ell_2}V(\ell_2)=\sqrt u\,a_v(u)v(z_0)+\varepsilon v(z_0)=(-\varepsilon+\varepsilon)v(z_0)=0$: for a newform the Hecke term and the Fricke
term cancel. By~\eqref{eq:mass}, $\Per_{\mathcal W_u}(v)=\sum_C|S_C|^{-1}\sum_{\ell_2}N(\ell_2)V_C(\ell_2)$ with $N$ as in Lemma~\ref{lem:period};
for inert $u$, $N\equiv1$ and the sum vanishes; for split $u$, $N=1-\mathbf 1[\text{isotropic}]$; for $u\parallel D$, $N(r)=u$ and
$N=1$ otherwise. The relation between $v$ and $v(u\,\cdot)$ is the Fricke involution of $\Gamma_0(u^2)$ acting on $\mathcal W_u$ as the
divisor pairing, since $v(-1/(u^2\tau))=\varepsilon v(u\tau)$. Numerically the split and ramified formulas reproduce the direct orbit
sums to all computed digits at $D=-8,-11,-20$ and $D=-3,-15$ ($u=3$). For the last sentence: the Fricke involution
$z\mapsto-1/(uz)$ of $\Gamma_0(u)$ maps the Heegner point of $[ua,b,c]$ to that of $[uc,-b,a]$, a bijection between the classes with
$b\equiv\beta$ and $b\equiv-\beta\pmod{2u}$ that preserves stabilisers. In the split case the isotropic lines give the two root classes,
which it exchanges, so the sum is $(1+\varepsilon)$ times the sum over one of them. In the ramified case it fixes the class
$b\equiv0$, so the sum equals $\varepsilon$ times itself. Numerically, for the even newform of level $5$ with $R=5.4362$ and
$\varepsilon=-1$ (LMFDB 5.0.1.5.1) the family period computed from the orbits of $\mathcal W_5$ is below $10^{-9}$ times the sum of the
absolute values of its terms at $D=-4,-11$ (split), $-15$ (ramified) and $-3,-8$ (inert); for the forms with $R=4.1324$ and $6.0540$
($\varepsilon=+1$) that ratio is between $0.03$ and $1$ in the split and ramified cases, and $\Per_{\mathcal W_5}(v(5\,\cdot))/\Per_{\mathcal W_5}(v)=+1$
(\texttt{newform-fricke-check.py}).
\end{proof}

\begin{remark}[the components of the family, and the inert identity explained]\label{rem:components}
Let $u$ be an odd prime with $u\nmid D$. By~\eqref{eq:mass} the piece over a class $C$ is a sum over the cosets $\gamma$ of the
seed at $\alpha\gamma^{-1}z_C$, weighted by $w_C(\gamma)=\mathbf 1[\ell_1\perp\ell_2]$. On the $u(u+1)$ ordered pairs of distinct lines, with the
uniform measure, write
\[
w_C=\bar w+\tfrac u{u-1}\bigl(n(\ell_1)+n(\ell_2)\bigr)+R_C,\qquad \bar w=\frac{u-\chi}{u(u+1)},\qquad n(\ell)=\frac{N(\ell)}u-\bar w,
\]
with $R_C$ orthogonal to every function of a single line. (i) Replacing $w_C$ by $\bar w$ sums the seed over all cosets, which
is Atkin's operator $U_u$ followed by the level-one Poincar\'e series; since the Fourier coefficients of the seed are homogeneous,
$\sum_{j\bmod u}\Psi^w_{Y'}((z+j)/u)=u^{-1}\Psi^w_{Y'}(z)$ exactly, and the trivial part of piece $u$ is $\bar w\,S^w_1(uY)$. It gives
the level-one lines the ratio $\bar w\,u^{-1/2}$ for every splitting type. (ii) By the proofs of Lemmas~\ref{lem:period}
and~\ref{lem:newperiod}, the values of $f_0,f_1,f_2$ and of the level-$u$ newforms at the coset points depend on at most one
line. So $R_C$ contributes no level-one and no level-$u$ line, and the marginal part carries $r_j(u;D)-\bar w\,u^{-1/2}$ and
all level-$u$ newform lines. (iii) For inert $u$ no line is isotropic, $N\equiv1=u\bar w$, the marginal part vanishes, and
$r_j(u;D)=\bar w\,u^{-1/2}=u^{-3/2}$. Arithmetically, piece $u$ is the restriction of the $u=1$ sum at length $uY$ to
$h'\equiv0\pmod u$; for a divisor $a$ of $h'^2-D$, $\ell_1$ is isotropic iff $u\mid a$ and $\ell_2$ iff $u\mid(h'^2-D)/a$. The three
parts are therefore computable from $\sigma_{-1}(h'^2-D)$ and $\sigma_{-1}((h'^2-D)/u)$, and they are tested on the data in
Section~\ref{sec:numerics}.
\end{remark}

\begin{proof}[Proof of Theorem~\ref{thm:level1} from the lemmas]
Theorem~\ref{thm:pieceu} restricted to the span of $f_0,f_1,f_2$ gives the coefficient $\sqrt u\sum_k\Per_{\mathcal W_u}(v_k)\tilde L(v_k)$
over an orthonormal basis $v_k$ of that span, which is the basis-free quantity $\sqrt u\,\mathbf P^{\mathsf T}\mathbf G^{-1}\mathbf L$
with $\mathbf P_m=\Per_{\mathcal W_u}(f_m)$, $\mathbf G_{mn}=\langle f_m,f_n\rangle$, $\mathbf L_m=\tilde L_{f_m}$. Dividing by
the $u=1$ coefficient $\Per_D(u_j)\tilde L_j/\|u_j\|^2$ and inserting the lemmas gives~\eqref{eq:r}; the phase $u^{-it_j}$
is from Theorem~\ref{thm:pieceu}. The asymptotics are not read off termwise --- $p_1\asymp u$ is large while $L_2=1/u$ and
$g\asymp\lambda u^{-1/2}$ are small, and the $\lambda_j(u)\sqrt u$ of $p_0$ cancels against $p_1(1/u-g)$ --- but from the closed
form~\eqref{eq:r} as a rational function of $\lambda$ and $\sqrt u$: with a computer algebra system, $r=u^{-3/2}$ identically
when $\chi=-1$, and the expansions in $u^{-1/2}$ stated in the theorem when $\chi=+1$ and $\chi=0$.
\end{proof}

\section{Numerical confirmation}\label{sec:numerics}

All data are the pieces of part~IV computed to $Y=10^7$ (\texttt{piece-divset.ts}, unrestricted object), sharp Riesz means
on $4096$ log-spaced points and smooth windows on $400$; amplitudes and phases by joint least squares at the level-one
lines with the dominant level-$u^2$ lines as nuisance frequencies (Str\"omberg's $\Gamma_0(9)$ eigenvalues at $u=3$
\cite{Stromberg2012}, the six strongest data-driven peaks at $u=5$). The predictions of Theorem~\ref{thm:level1} were
computed twice: from~\eqref{eq:r}, and directly by evaluating the Maass form at the orbit representatives of $\mathcal W_u$,
integrating the Gram matrix over the cosets of $\Gamma_0(u^2)$, and orthonormalising; the two agree to four digits, and
the direct route also checks $\langle f_0,f_0\rangle/\|u_j\|^2=u(u+1)$ and the Iwaniec--Luo--Sarnak entry.

\paragraph{The case $u=2$.} The local constants are not integers ($p_1=4.913$, $5.278$, $0.321$, $1.373$ for
$D/4\equiv2,3\ (4)$, $D/4\equiv1\ (4)$, $D\equiv1\ (8)$, $D\equiv5\ (8)$): the multiplicity depends on the class through
$B_C$ modulo $2$, Proposition~\ref{prop:classwise}, and~\eqref{eq:r} applies with the class-by-class $P$. The values so obtained
coincide with the direct computation and match the data (Table~\ref{tab:ratios}).

\paragraph{Composite $u$.} For squarefree $u=u_1u_2$ the oldform space is $\{u_j(dz):d\mid u^2\}$, of dimension $9$ for $u=6$,
and Proposition~\ref{prop:mult} gives $r_j(u;D)=\prod_{p\mid u}r_j(p;D)$. The direct computation at level $36$ (Gram matrix over the
$72$ cosets, $\|f_1\|^2/\|u_j\|^2=72.000$) was made before the proposition was proved and confirms it: $r_1(6;-12)=+0.6573$ and
$r_1(6;-8)=-0.2642$, against the products $r_1(2;-12)\,r_1(3;-12)=0.945\times0.695=0.657$ and
$r_1(2;-8)\,r_1(3;-8)=0.850\times(-0.311)=-0.264$; and the $9\times9$ Gram matrix is the tensor product of the $3\times3$
matrices at levels $4$ and $9$ entry by entry ($0.0781=0.7303\times0.1069$, $-0.2323=0.7303\times(-0.3181)$,
$-0.0955=0.3001\times(-0.3181)$, \dots). The data at $u=6$ are noisy (the level-$36$ spectrum is dense): the five signs observed
agree with the products and the magnitudes scatter by up to $0.2$ ($D=-8,-20,-15,-4,-12$: observed
$-0.449,-0.320,-0.295,+0.111,+0.549$ against $-0.264,-0.264,-0.237,+0.163,+0.657$).

\paragraph{The cycloidal lines.} At $u=3$ the pieces are dominated by the lines $3.536$, $5.504$, $6.647$, $7.432$, $8.698$,
which Str\"omberg attributes to forms of the cycloidal group $\Gamma^3$~\cite{Stromberg2012}; their coefficients are not
tabulated and their amplitudes in piece $3$ cannot be predicted here.

\paragraph{The components of the piece.} Remark~\ref{rem:components} was tested on the same grids for $u=3$ ($D=-8,-11,-20$ split,
$-4,-7$ inert) and $u=5$ ($D=-4,-11$ split, $-8$ inert), with the three parts computed from $\sigma_{-1}(h'^2-D)$ and
$\sigma_{-1}((h'^2-D)/u)$ (\texttt{hecke-components.ts}); their sum reproduces the grids of part~IV. For inert $u$ the marginal part
vanishes identically. At $u=3$ the level-$3$ newform line $5.0987$ lies entirely in the marginal part: amplitudes $0.143$, $0.138$,
$0.191$ there against $0.141$, $0.136$, $0.192$ in the piece, and at most $0.007$, at the noise, in $R$. The cycloidal lines $3.536$
and $6.647$ lie entirely in $R$ (for example $0.2009$ against $0.2010$ at $D=-4$), with at most $0.008$ in the marginal part. The
trivial part carries $t_1$ with the ratio $\bar w\,u^{-1/2}$: observed $+0.100$, $+0.090$, $+0.083$ against $+0.096$ (split) and
$+0.193$, $+0.188$ against $+0.192$ (inert). In the marginal part and in $R$ the line $t_1$ is at or below the noise of these grids,
which is consistent with the prediction but does not test it. At $u=5$ the marginal part carries the even level-$5$ newforms
$4.1324$, $6.0540$ and $8.2947$, all with $\varepsilon=+1$ in the LMFDB, at $9$ to $27$ times its noise for $D=-4$ and $4$ to $12$ times
for $D=-11$; the forms $5.4362$ and $7.9865$ with $\varepsilon=-1$ are absent from it, as Lemma~\ref{lem:newperiod} requires. $R$
carries lines at $2.64$--$2.71$, at $5.22$--$5.28$ and near $10.8$, none at a level-$5$ newform. By size the trivial part is negligible
(root mean square $0.007$--$0.022$) against the marginal part ($0.08$--$0.17$) and $R$ ($0.15$--$0.32$): what is not uniform in $u$
lives at the levels $u$ and $u^2$.

\paragraph{Precision of the subtracted means.} The means $\mathbb E\sigma^*_{-1}$ subtracted in all grids are floating-point Euler
products over $10^6$ primes or more. Against a log-sum their rounding and tail errors lie between $10^{-12}$ and $4\cdot10^{-11}$. An
error $\delta$ adds about $-0.111\,\delta\,Y^{3/2}$ to $\sqrt Y\,S^w_u(Y)$, up to $0.16$ at $Y=10^7$, and polynomial detrending in $\log Y$
does not remove it; the components above were fitted with a $Y^{3/2}$ term. Refitting Table~\ref{tab:ratios} with such terms moves
the $u=2$ entries by at most $0.015$, the $u=3$ entries by at most $0.04$ (split $-0.315$, $-0.353$, $-0.354$; ramified $+0.618$,
$+0.707$; inert $+0.177$) and the $u=5$ ramified entry to $+0.445$. The amplitudes of Table~\ref{tab:newform} move by at most $2\%$
and the phases by at most $0.02$ rad. The tables keep the original fits, and the agreement at $u=3$ is to be read with this
systematic uncertainty.

\section{What is open}\label{sec:open}
\begin{enumerate}
\item Closed forms for the
class-dependent entries of Proposition~\ref{prop:classwise} (at $u=2$, and for $u^2\mid D$) in terms of $\lambda_j(u)$ and the
periods of the discriminants $D/u^{2k}$, via the Hecke relations.
\item The lines of the cycloidal-group forms, and any bound for the newform lines uniform in the newform: the level-aspect
equidistribution of \cite{LiuMasriYoung2013,HumphriesNordentoft2025}, route 1b of the knowledge base.
\item The error term uniform in $u$ (part~IV, Remark~\textsc{gaps}), which is the conjecture of part~III itself.
\end{enumerate}
